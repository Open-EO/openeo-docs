javascript
import openeo # 1. Connect to a back-end conn = openeo.connect("openeo.cloud") conn.authenticate_oidc() # 2. Load a collection cube = conn.load_collection( "SENTINEL2_L2A", spatial_extent={"west": 4.0, "east": 4.5, "south": 51.0, "north": 51.5}, temporal_extent=["2024-06-01", "2024-08-31"], bands=["B04", "B08"] ) # 3. Apply a process (NDVI) ndvi = cube.ndvi(nir="B08", red="B04") # 4. Download the result ndvi.download("ndvi_output.nc", format="NetCDF")
